import React, { useState, useEffect } from 'react';

  pressure: number;

interface Metric {
	label: string
	value: string | number
	unit?: string

const PumpDashboard = () => {
  const [data, setData] = useState<any>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const fetchData = async () => {
    try {
      const response = await fetch('http://145.249.191.141:15001/WinCCRestService/tagManagement/values');
      if (!response.ok) throw new Error('Ошибка сети или сервера');
      const result = await response.json();
      setData(result);
      setError(null);
    } catch (err) {
      setError('Не удалось получить данные с сервера');
      console.error(err);
    } finally {
      setLoading(false);
    }
  };

  // Выполняем запрос при монтировании и каждые 5 секунд
  useEffect(() => {
    fetchData(); // Первый запрос сразу
    const interval = setInterval(fetchData, 5000); 
    return () => clearInterval(interval); // Очистка при размонтировании
  }, []);

  if (loading) return <div className="p-10 text-slate-400">Загрузка данных...</div>;
  if (error) return <div className="p-10 text-red-500">{error}</div>;

  return (
    <div className="min-h-screen bg-slate-950 p-6 text-slate-100 font-sans">
      <h1 className="text-2xl font-bold mb-6 text-white">Мониторинг (Данные WinCC)</h1>
      
      {/* Пример вывода данных */}
      <div className="bg-slate-900/50 border border-slate-800 rounded-xl p-6">
        {/* Здесь вы используете data для заполнения таблицы */}
        <pre className="text-sm font-mono text-emerald-400">
           {JSON.stringify(data, null, 2)}
        </pre>
      </div>
    </div>
  );
};

export default PumpDashboard;
